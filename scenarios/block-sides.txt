# Drop low and look across the hills, so the side faces of the terraces
# (grass hanging over dirt) are big enough to judge. Pair with
# texture-distances.txt, which mostly shows top faces.
# Run: cargo headless scenarios/block-sides.txt
camera 64 56 92 0 -37.9            # the start's view, flying (the player starts on foot)
wait 5
hold MouseLeft 1   # lock the cursor so the mouse turns the camera
mouse 0 -250       # level the view out, just below the horizon
hold ShiftLeft 110 # sink toward the hilltops
hold KeyW 60       # move up to the nearest terraces
wait 1
screenshot target/scenarios/block-sides/low.png
inspect FlyCamera
