# The hotbar and the inventory: switch slots with the wheel, open the
# inventory, take items from the catalog, put one on the hotbar, and
# place it.
# Run: cargo headless scenarios/inventory.txt
camera 64 56 92 0 -37.9          # the start's view, flying
fill 61 48 86 67 58 86 planks    # a wall in reach, to place against
wait 5
hold MouseLeft 1                 # lock the cursor
wait 1
screenshot target/scenarios/inventory/0-hotbar.png
wait 1

wait 1
inspect ActiveSlot
wait 1

hold KeyE 1                      # open the inventory
wait 3

wait 1
hold MouseLeft 1
wait 1
release ShiftLeft
wait 1
hold MouseLeft 1                 # and one more
wait 2
screenshot target/scenarios/inventory/2-taken.png
wait 1

wait 1
cursor 700 500
wait 1
cursor 880 676
wait 2
screenshot target/scenarios/inventory/3-dragging.png

wait 2

wait 3
