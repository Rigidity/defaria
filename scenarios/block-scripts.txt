# Scripted blocks: a stone wall with one mossy block at its corner, and
# three switched lamps in front of it, on a stone platform. Right click
# on the middle lamp turns it on (its script's `on_used`), without
# placing anything against it, and `inspect BlockData` shows it counted
# the switch. Then random ticks are sped up (`random-ticks`), and the
# moss creeps over the wall's bare stone (`on_random_tick`), but not onto
# the iron ore in it.
# Run: cargo headless scenarios/block-scripts.txt
camera 28 50.5 38.5 0 -25                                 # south of it, looking north
settle                                                    # so it's loaded, whatever the radius
wait 5
fill 20 45 20 40 45 36 stone                              # the platform
fill 21 46 20 39 52 36 air                                # cleared above it
fill 22 46 26 34 49 26 stone                              # the wall
fill 26 47 26 27 48 26 stone[ore=defaria:iron]            # iron ore in it
fill 22 46 26 22 46 26 mossy_stone                        # moss at its corner
fill 25 46 30 25 46 30 switch_lamp                        # three lamps, off
fill 28 46 30 28 46 30 switch_lamp
fill 31 46 30 31 46 30 switch_lamp[lit=true]              # but this one
wait 2
screenshot target/scenarios/block-scripts/0-built.png
wait 1
hold MouseLeft 1                                          # lock the cursor
camera 28.5 48.5 31.5 0 -60                               # over the middle lamp
wait 1
hold MouseRight 1                                         # use it
wait 2
inspect BlockData
camera 28 50.5 38.5 0 -25
wait 2
screenshot target/scenarios/block-scripts/1-switched.png
wait 1
random-ticks 256                                          # moss spreads fast
wait 240
random-ticks 1
wait 1
screenshot target/scenarios/block-scripts/2-mossy.png
